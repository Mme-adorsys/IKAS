    Step 3 — Create agentshield/src/data/cve-lookup.ts:

    ```typescript
    import { randomUUID } from 'crypto';
    import { DiscoveredServer } from '../types/discovery';
    import { Finding, SeverityLevel } from '../types/findings';

    interface CveTemplate {
      cveId?: string;
      owaspCategory: string;
      title: string;
      severity: SeverityLevel;
      score: number;
      description: string;
      remediation: string;
    }

    interface CveTableEntry {
      id: string;            // stable identifier for the row, e.g. 'CVE-2025-49596'
      match: (server: DiscoveredServer) => boolean;
      build: (server: DiscoveredServer) => CveTemplate;
    }

    const COMMAND_INJECTION_TOOL_NAMES = ['create-user', 'delete-user', 'write_neo4j_cypher'];

    export const CVE_LOOKUP_TABLE: CveTableEntry[] = [
      {
        id: 'CVE-2025-49596',
        match: (s) => s.baseUrl.includes('localhost') || s.baseUrl.includes('127.0.0.1'),
        build: (_s) => ({
          cveId: 'CVE-2025-49596',
          owaspCategory: 'MCP07:2025',
          title: 'Localhost-bound MCP server — CVE-2025-49596 DNS-rebinding attack surface',
          severity: 'high',
          score: 9.4,
          description:
            'MCP server bound to localhost is in the attack surface of CVE-2025-49596 (CVSS 9.4): ' +
            'the MCP Inspector proxy on 6274/6277 does not validate Origin headers, enabling ' +
            'DNS-rebinding/CSRF RCE from a malicious website visited by the developer.',
          remediation:
            'Bind the MCP server to a non-loopback interface and place it behind authenticated network ' +
            'access; upgrade MCP Inspector to a version that validates Origin headers.',
        }),
      },
      {
        id: 'CVE-2025-6514',
        match: (s) => s.transport === 'mcp-jsonrpc' && s.hasAuth === false,
        build: (_s) => ({
          cveId: 'CVE-2025-6514',
          owaspCategory: 'MCP07:2025',
          title: 'Unauthenticated MCP JSON-RPC server — CVE-2025-6514 surface exposed',
          severity: 'high',
          score: 9.6,
          description:
            'Unauthenticated MCP JSON-RPC server is reachable. CVE-2025-6514 (CVSS 9.6) allows OS ' +
            'command injection through OAuth authorization_endpoint manipulation in mcp-remote clients. ' +
            'Any unauthenticated MCP server can be impersonated to deliver malicious OAuth redirects.',
          remediation:
            'Require authentication on the MCP endpoint (Bearer token or mTLS); restrict outbound ' +
            'mcp-remote clients to a vetted allow-list of authorization_endpoint hosts.',
        }),
      },
      {
        id: 'KEYCLOAK-REST-NOAUTH',
        match: (s) => s.transport === 'rest-keycloak',
        build: (_s) => ({
          owaspCategory: 'MCP07:2025',
          title: 'Keycloak MCP REST server — no authentication or rate limiting on tool endpoints',
          severity: 'high',
          score: 7.5,
          description:
            'The Keycloak MCP REST server exposes all tool endpoints (e.g. /tools/list-users, ' +
            '/tools/create-user) without authentication or rate limiting. Documented in IKAS CONCERNS.md.',
          remediation:
            'Add authentication middleware (Bearer token or Keycloak service account) and rate limiting ' +
            '(express-rate-limit) on /tools/* routes.',
        }),
      },
      {
        id: 'TOOL-COMMAND-INJECTION',
        match: (s) => s.tools.some((t) => COMMAND_INJECTION_TOOL_NAMES.includes(t.name)),
        build: (s) => {
          const matched = s.tools
            .map((t) => t.name)
            .filter((n) => COMMAND_INJECTION_TOOL_NAMES.includes(n))
            .join(', ');
          return {
            owaspCategory: 'MCP05:2025',
            title: 'MCP server exposes write/exec tool without explicit input-validation contract',
            severity: 'high',
            score: 8.1,
            description:
              `Tool(s) ${matched} accept arguments that flow into Keycloak Admin API or Neo4j Cypher ` +
              'without an input-validation schema. Documented IKAS vulnerability: MCP tool argument ' +
              'injection possible.',
            remediation:
              'Add zod / JSON-schema validation at the tool boundary; sanitize Cypher inputs by ' +
              'rejecting embedded statement separators; reject identifiers that do not match ' +
              '/^[A-Za-z0-9_-]+$/.',
          };
        },
      },
    ];

    export function applyCveLookup(servers: DiscoveredServer[]): Finding[] {
      const findings: Finding[] = [];
      for (const server of servers) {
        for (const entry of CVE_LOOKUP_TABLE) {
          if (!entry.match(server)) continue;
          const tpl = entry.build(server);
          findings.push({
            id: randomUUID(),
            title: tpl.title,
            description: tpl.description,
            severity: tpl.severity,
            component: server.baseUrl,
            score: tpl.score,
            remediation: tpl.remediation,
            owaspCategory: tpl.owaspCategory,
            ...(tpl.cveId ? { cveId: tpl.cveId } : {}),
          });
        }
      }
      return findings;
    }
    ```
